    \column{0.52\textwidth}
    \begin{block}{Image Formation Model}
        \vspace{-0.15cm}
        \begin{equation*}
            y = \mathcal{T}(x * k) + n
        \end{equation*}
        \vspace{-0.35cm}
        {\scriptsize
        \begin{itemize}
            \setlength{\itemsep}{1pt}
            \setlength{\parskip}{0pt}
            \item[$x$] Latent sharp scene image.
            \item[$k$] Blur kernel (PSF).
            \item[$*$] Convolution operation.
            \item[$\mathcal{T}$] Camera response / intensity transform.
            \item[$n$] Sensor noise.
            \item[$y$] Observed degraded image.
        \end{itemize}}
    \end{block}
